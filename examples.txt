;; stac examples for the REPL
;;
;; Start stac with the prelude, which imports SC's UGens and the standard library:
;;
;;     build/stac -p prelude.stac
;;
;; Paste one line at a time. Each line stands alone, or uses what a line above it in its
;; section bound. The REPL prints the stack after each line; `clear` empties it, and `stop`
;; silences everything. Tab completes a name; the hint after a word shows what it takes and
;; leaves. Every word is in docs/dictionary/; docs/phrasebook.md sets sclang beside stac.
;; tests/py/test_examples_txt.py runs this file through the REPL.


;;;; 1. First sounds

[440 sinosc .1 *] play                              ; a sine, 440 Hz
stop clear
[[440 442] sinosc .1 *] play                        ; two channels: a list of frequencies
stop clear
[440 sinosc .1 *] play = n                          ; keep the synth's node
n.free                                              ; and free that one synth
[[1 2 3 4] 200 * sinosc [1 .5 .33 .25] * +/ .05 *] play   ; 4 partials, summed with +/
stop clear


;;;; 2. Oscillators and noise

[110 saw .05 *] play
stop clear
[110 .3 pulse .05 *] play                           ; pulse width .3
stop clear
[pinknoise .05 *] play
stop clear
[[100 100.5 99.7 200.3] saw [-1 -.33 .33 1] pan2 +/ .03 *] play   ; 4 voices across the stereo field
stop clear


;;;; 3. Filters and LFOs

[110 saw 800 .2 rlpf .1 *] play                     ; resonant low-pass at 800 Hz
stop clear
[110 saw  2 lfnoise1.kr 300 3000 bilin  .2 rlpf  .1 *] play   ; the cutoff wanders, 2 Hz
stop clear
[whitenoise  .5 sinosc.kr 400 4000 biexp  .1 bpf  .3 *] play   ; a swept band of noise
stop clear
[2 lfnoise1.kr.poll 200 * 400 + sinosc .1 *] play   ; .poll prints the LFO on scsynth's console
stop clear


;;;; 4. SynthDefs, synths and controls

synthdef ping freq:440 amp:.1 gate:1 [freq sinosc .005 .2 env.asr gate envgen.kr.done * amp * 2 X]
synthdef pad freq:220 amp:.1 gate:1 [freq saw 2000 .3 rlpf .01 .2 .6 1 env.adsr gate envgen.kr.done * amp * 2 X]
{:freq 660} ping = x                                ; start a synth with a control value
{:freq 880} x.set                                   ; change it while it plays
x.release                                           ; release its envelope
{:freq 110} pad = p
{:freq 165 :amp .05} p.set
4 p.releaseIn                                       ; release over 4 seconds
synthdef click freq:1320 [freq sinosc .002 .3 env.perc 1 envgen.kr.done * .2 * 2 X]
{} click                                            ; frees itself: .done on its envelope
{:freq 2000} click
clear


;;;; 5. Effects

synthdef verb bus mix:.4 [bus 2 in mix .8 .5 freeverb]
2 bus.audio = fx                                    ; a private stereo bus
{:bus fx :addAction 'tail} verb = v                 ; the reverb runs after its sources
{:out fx :freq 330} ping = a                        ; the source writes to the bus
a.release
{:out fx :freq 990} click
v.free
clear
[2 dust .01 .2 decay2 800 sinosc * = s  s s .5 .375 4 combc + .2 *] play   ; clicks and their echo
stop clear
[100 saw 8 * tanh .05 *] play                       ; tanh distortion
stop clear


;;;; 6. Patterns

{:instrument 'ping :midinote [60 62 64 67] cyc :dur .25} pbind play = mel
mel.stop
{:instrument 'ping :degree [0 2 4 5 7] picks :scale 'minor :dur .125} pbind play = r   ; random notes of a scale
r.stop
{:instrument 'pad :midinote [[60 64 67] [62 65 69]] cyc :dur 1 :legato .9} pbind play = ch   ; chords
ch.stop
{:instrument 'ping :midinote [60 62 64] cyc :dur .25} pbind 'tune pdef = t   ; a pattern you can replace
{:instrument 'ping :midinote [67 65 64 62] cyc :dur .125} pbind 'tune pdef   ; replaces it on the next beat
t.stop
2 clock.new = c                                     ; 2 beats per second
{:instrument 'ping :midinote [72 67] cyc :dur 1} pbind c.play = q
{:tempo 4} c.set                                    ; faster, while it plays
q.stop


;;;; 7. Sounds you replace while they play

[[55 55.2] saw .05 *] 'drone ndef = d
[[55 55.2] saw 600 .3 rlpf .05 *] 'drone ndef       ; crossfades into the new sound
{:fadeTime 3} d.set
[[82.5 82.8] saw 900 .3 rlpf .05 *] 'drone ndef     ; a 3 s crossfade
d.stop                                              ; silent, but still running
[d.ar  .2 lfnoise1.kr 200 2000 biexp  .2 rlpf] 'shaped ndef = sh   ; filter the drone in another ndef
sh.end
d.end
clear


;;;; 8. Sequencing inside a synth

[8 impulse.kr 0 [60 63 67 70] inf dseq demand.kr nnhz saw .05 *] play   ; a step sequence, 8 steps a second
stop clear
[6 impulse.kr 0 48 72 inf diwhite demand.kr nnhz sinosc .1 *] play      ; random MIDI notes, 48 to 72
stop clear


;;;; 9. The language without sound

1 2 +                                               ; the REPL prints the stack
clear
ord 5 take                                          ; 1 2 3 4 5, from an infinite list
[1 2 3] 10 *
[1 2 3] [10 20 30] +
[1 2 3 4] +/                                        ; sum
[1 2 3 4] +\                                        ; running sum
[3 1 2] sort
[1 2] @ [10 20] +                                   ; each with each
60 nnhz                                             ; MIDI note to Hz
440 hznn
clear
\x [x x *] = square                                 ; a function, bound to a name
5 square
def cube x --> num "x cubed." [x x x * *]           ; a def checks what it leaves
3 cube
`cube help
clear


;;;; 10. Live editing

def detune f --> num [f 1.01 *]
{:instrument \f [f f detune 2ple sinosc .01 .4 env.perc 1 envgen.kr.done * .05 *] :f [220 330] cyc :dur .5} pbind play = live
def detune f --> num [f 1.05 *]                     ; the next note uses the new detune
live.stop
stop clear


;;;; 11. Files

[440 sinosc .1 *] .5 "tone" render                  ; tone.wav in $STAC_RECORDINGS
{:instrument 'ping :midinote [60 64 67] cyc :dur .25} pbind 2 "arpeggio" render
`ping.write                                         ; ping.scsyndef in $STAC_SYNTHDEFS
